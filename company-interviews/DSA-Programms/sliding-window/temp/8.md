// Find the missing number in an array of 1 to n.

package main

import "fmt"

func main() {
	res := 0
	res1 := 0
	n := 10
	arr := []int{1, 2, 3, 4, 5, 6, 7, 8, 10}

	// XOR all numbers from 1 to n
	for i := 1; i <= n; i++ {
		res ^= i
	}

	// XOR all elements in the array
	for _, num := range arr {
		res1 ^= num
	}

	result := res ^ res1

	fmt.Println(result)
}
