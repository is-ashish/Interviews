// 7. Find the subarray with sum equal to k.
package main

import (
	"fmt"
)

func main() {

	arr := []int{1, 2, 3, 7, 5}
	k := 17
	prefixMap := make(map[int]int)
	sum := 0

	for i := 0; i < len(arr); i++ {
		sum = sum + arr[i]
		target := sum - k
		if index, exist := prefixMap[target]; exist {
			fmt.Println("value found", arr[index+1:i+1])
			return
		}

		prefixMap[sum] = i
	}

	fmt.Println("No target sub array found", prefixMap)

}
