package main

import "fmt"

func main() {
	arr := []int{2, 5, 1, 7, 3, 4, 6, 2}
	k := 14
	sum := 0
	maxCount := 0
	s := 0 // start pointer

	for i := 0; i < len(arr); i++ {
		// 1. Expand the window
		sum += arr[i]
		// 2. Shrink the window if it violates the rule (sum > k)
		if sum > k && s <= i {
			sum -= arr[s]
			s++
		}
		// 3. NOW the window [s:i] is guaranteed to be <= k.
		// Update maxCount every single time.
		currentWindowSize := i - s + 1
		if currentWindowSize > maxCount {
			maxCount = currentWindowSize
		}
	}

	fmt.Println("Longest subarray length less than or equal to k is:", maxCount)
}
