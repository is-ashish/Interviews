// Move Zeroes to End: Rearrange an array so all zeroes are at the end while maintaining order of non-zero elements.
package main

import "fmt"

func main__() {
	arr := []int{0, 0, 0, 5, 0, 0, 9, 0, 0, 0, 3, 0, 0, 1} // o/p = [4,5,6,0,0,0]
	lastNonZeroIndex := 0

	for i := 0; i < len(arr); i++ {
		if arr[i] != 0 {
			arr[lastNonZeroIndex] = arr[i]
			lastNonZeroIndex++
		}
	}
	for i := lastNonZeroIndex; i < len(arr); i++ {
		arr[i] = 0
	}
	fmt.Println(arr)
}
