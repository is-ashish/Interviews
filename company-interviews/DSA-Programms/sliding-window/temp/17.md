// Number of Substrings Containing All Three Characters

func numberOfSubstrings(s string) int {
	count := 0
	lastSeen := [3]int{-1, -1, -1}
	for i := 0; i < len(s); i++ {
		lastSeen[s[i]-'a'] = i

		if lastSeen[0] != -1 && lastSeen[0] != -1 && lastSeen[0] != -1 {
			count = count + (1 + getMin(lastSeen[0], lastSeen[1], lastSeen[2]))
		}

	}
	return count
}

func getMin(a, b, c int) int {
	if a <= b && a <= c {
		return a
	} else if b <= a && b <= c {
		return b
	} else {
		return c
	}
}
