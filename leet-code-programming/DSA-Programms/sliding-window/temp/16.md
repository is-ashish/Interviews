// maximum amount of water a container can store.

package main

import "fmt"

func main() {
	height := []int{1, 8, 6, 2, 5, 4, 8, 3, 7}
	l := 0
	r := len(height) - 1
	maxArea := 0

	for i := 0; i < len(height)-1; i++ {
		h := getHeight(height[l], height[r])
		w := height[r] - height[l]

		area := h * w

		if maxArea < area {
			maxArea = area
		}

		if height[l] > height[r] {
			l++
		} else {
			r--
		}
	}

	fmt.Println(maxArea)

}

func getHeight(a, b int) int {
	if a < b {
		return a
	} else {
		return b
	}
}
