// Find the max fruit into basket of type k
package main

import "fmt"

func main() {
	arr := []int{3, 3, 3, 1, 2, 1, 1, 2, 3, 3, 4}
	k := 3
	l := 0
	maxTree := 0

	mapCount := make(map[int]int)

	for i := 0; i < len(arr); i++ {
		mapCount[arr[i]]++
		for len(mapCount) > k {
			mapCount[arr[l]]--
			if mapCount[arr[l]] == 0 {
				delete(mapCount, arr[l])
			}
			l++
		}
		if maxTree < i-l+1 {
			maxTree = i - l + 1
		}
	}
	fmt.Println(maxTree) // Output: 5
}
