// Find the maximum subarray sum (Kadane’s Algorithm).

package main

import (
	"fmt"
)

func main() {
	arr := []int{-5, -, -8, -2}
	maxSum := arr[0]
	currSum := 0
	start := 0
	end := 0
	res := []int{}

	for i := 1; i < len(arr); i++ {
		if currSum < 0 {
			currSum = 0
			start = i
		}
		currSum = currSum + arr[i]

		if maxSum < currSum {
			maxSum = currSum
			end = i
		}
	}

	for j := start; j <= end; j++ {
		res = append(res, arr[j])
	}

	fmt.Println("Max sub array value", maxSum)
	fmt.Println("Sub array elements", res)

}

// The trick is updating the start pointer whenever currSum resets to 0, and capturing the end pointer whenever you find a new maxSum.
