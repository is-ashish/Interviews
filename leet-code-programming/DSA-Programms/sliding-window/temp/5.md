// • Leaders in Array: Find elements that are greater than all elements to their right.

package main

import (
	"fmt"
)

func main_() {
	arr := []int{16, 17, 4, 13, 5, 2}
	res := []int{}

	if len(arr) == 0 {
		fmt.Println(res)
	}
	currentMaxIndex := len(arr) - 1
	for i := len(arr) - 1; i > 0; i-- {
		if arr[i] >= arr[currentMaxIndex] {
			res = append(res, arr[i])
			currentMaxIndex = i
		}

	}
	fmt.Println(res)
}
