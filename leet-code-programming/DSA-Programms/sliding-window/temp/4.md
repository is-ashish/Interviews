//• Second Largest Element: Find the second largest number in an array.

package main

import (
	"fmt"
	"math"
)

func main__() {
	arr := []int{5, 7, 12, 8, 6, 19} // o/p = [4,5,6,0,0,0]
	first := math.MinInt
	second := math.MinInt
	for _, num := range arr {
		if num > first {
			second = first
			first = num
		} else if num > second {
			second = num
		}
	}
	fmt.Println(second, first)

}

func max(a, b int) int {
	if a > b {
		return a
	} else {
		return b
	}
}
