// 14. Find the maximum length of substring with all distinct elements.
package main

import "fmt"

func main_() {

	str := "cadbbzabcdefgh"
	maxCount := 0
	l := 0

	// charMap := make(map[byte]int)
	charIndex := [256]int{}

	for i := 0; i < len(str); i++ {
		if charIndex[str[i]] > 1 && charIndex[str[i]] > l {
			l = charIndex[str[i]] + 1
		}
		charIndex[str[i]] = i + 1
		curLen := i - l + 1
		if maxCount < curLen {
			maxCount = curLen
		}
	}
	fmt.Println(maxCount)

}
